/* Method 1: array of row pointers */
int **m = malloc(r * sizeof(int *));
for (int i = 0; i < r; i++)
    m[i] = malloc(c * sizeof(int));
m[2][3] = 7;
for (int i = 0; i < r; i++) free(m[i]);   /* free rows FIRST */
free(m);
/* Method 2: one contiguous block (cache/DMA friendly) */
int *flat = malloc(r * c * sizeof(int));
#define M(i,j) flat[(i)*c + (j)]
M(2,3) = 7;
free(flat);
